cmake_minimum_required(VERSION 3.15)
project(HobbyTracker LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

find_package(imgui CONFIG REQUIRED)
find_package(glfw3 CONFIG REQUIRED)

add_executable(HobbyTracker main.cpp)

target_link_libraries(HobbyTracker PRIVATE 
    imgui::imgui 
    glfw
    opengl32  # ← Добавляем системную библиотеку OpenGL для Windows
)
